\begin{abstract}
Current Large Language Models (LLMs) process continuous activations within untyped vector spaces, rendering them vulnerable to prompt injection attacks and unauthorized activation probes. We introduce \textbf{Neural State Architecture (NSA)}, a computational paradigm that embeds typed activations $h = (m, \sigma_h, \sigma_s, \nu)$ and exact algebraic transition operators $V \in \mathcal{T}_\Sigma$ directly into neural forward passes. We rigorously distinguish between \textbf{Tier 1 Structural Enforcement} (hard attention non-interference $A_{ij} = 0$, exact transition projection, and capability-gated security execution automata) and \textbf{Tier 2 Statistical Monitoring} (speculative multi-layer residual probing over checkpoint coverage $\mathcal{L}_A$). We prove whole-network non-interference via observational equivalence ($x \equiv_L x' \implies F(x) \equiv_L F(x')$) and validate NSA on injection benchmarks, demonstrating a $0.00\%$ hijack rate when structural constraints compose with value alignment.
\end{abstract}

\section{Introduction}
Modern LLMs operate on untyped continuous spaces $\mathbb{R}^d$. This architecture fails to provide mathematical guarantees regarding information boundaries.

\begin{definition}[Typed Quad-Tuple Manifold]
Each token hidden state at layer $l$ is represented as an authoritative quad-tuple:
\begin{equation}
h_i^{(l)} = \left( m_i^{(l)}, \sigma_{h,i}^{(l)}, \sigma_{s,i}^{(l)}, \nu_i^{(l)} \right) \in \mathbb{R}^d \times \Sigma_h \times \Sigma_s \times \mathcal{V}
\end{equation}
where $m_i$ is the semantic activation, $\sigma_h \in \Sigma_C \times \Sigma_I \times \Sigma_A \times \Sigma_L$ is the hard trusted policy state, $\sigma_s \in \Sigma_U \times \Sigma_R$ is the soft operational risk state, and $\nu \in \mathcal{V}$ is the value/preference alignment state.
\end{definition}

\section{Exact Transition Projection \& Policy Semantics}
Edges carry paired operators $(w, V)$ where $V \in \mathcal{T}_\Sigma$ is projected onto the cone of legal transitions:
\begin{equation}
\mathcal{P}_{\mathcal{T}_\Sigma}(V) = \text{triu}(V) - \text{diag}(\text{diag}(V)) + \text{diag}(\max(0, \text{diag}(V)))
\end{equation}
rendering unauthorized downward declassifications unrepresentable by construction.

State-aware attention enforces policy via:
\begin{equation}
A_{ij} = \text{Softmax}\left( \frac{Q_i K_j^T}{\sqrt{d_k}} + \mathbf{M}(\boldsymbol{\sigma})_{ij} \right)
\end{equation}
where $\mathbf{M}(\boldsymbol{\sigma})_{ij} = -\infty$ whenever $\sigma_{h,i} < \sigma_{h,j}$ (Hard Policy Semantics).

\section{Observational Equivalence \& Non-Interference}

\begin{definition}[Low-Equivalence $\equiv_L$]
Let $L \in \Sigma_h$ denote an observer's clearance. Two input sequences $X, X'$ are low-equivalent ($X \equiv_L X'$) if and only if:
\begin{equation}
\forall t : \sigma_{h,t} \le L \implies X_t = X'_t
\end{equation}
\end{definition}

\begin{theorem}[Whole-Network Structural Non-Interference]
Let $F$ be an $N$-layer NSA network with hard state masking and exact transition operators $V \in \mathcal{T}_\Sigma$. For any observer clearance $L$:
\begin{equation}
X \equiv_L X' \implies F(X) \equiv_L F(X')
\end{equation}
\end{theorem}

\begin{proof}
For any layer $l$ and query position $i$ with $\sigma_{h,i} \le L$, attention weights $A_{ij}^{(l)} = 0$ for all $j$ where $\sigma_{h,j} \not\le \sigma_{h,i}$. Hence, the output activation $v_{\text{out}, i}^{(l)}$ depends exclusively on coordinates with $\sigma_{h,j} \le L$, which are identical between $X$ and $X'$. By induction on depth, $F(X)_i = F(X')_i$ for all positions observable at clearance $L$.
\end{proof}

\section{Two-Tier Defense \& NSA 2.0 Runtime}

\subsection{Privilege Escalation Prevention}
\textbf{Axiom}: \textit{Semantic content may not manufacture hard authority} ($m_t \not\to \sigma_{h, t+1}$). We define the Security Execution Automaton $(Q, \Sigma_h, \Sigma_s, \mathcal{C}, \delta)$. Transitions into $SYSTEM$ state require a verified external capability $c_t \in \mathcal{C}$:
\begin{equation}
(q_t, \sigma_t, c_t) \xrightarrow{\delta} (q_{t+1}, \sigma_{t+1}) \iff \text{Authorized}(c_t, q_t, q_{t+1}) = 1
\end{equation}

\subsection{Statistical Checkpoint Monitoring}

Because $P(\hat{\sigma} = \sigma) < 1$, the auditor operates as an empirical anomaly detector triggering early-exit KV rollback upon detecting potential out-of-distribution transitions.
