\begin{abstract}
Current Large Language Models (LLMs) process continuous activations within untyped vector spaces, rendering them vulnerable to prompt injection attacks and unauthorized activation probes. We introduce \textbf{Neural State Architecture (NSA)}, a computational paradigm that embeds typed activations $h = (m, \sigma_h, \sigma_s, \nu)$ and exact algebraic transition operators $V \in \mathcal{T}_\Sigma$ directly into neural forward passes. We rigorously distinguish between \textbf{Tier 1 Structural Enforcement} (hard attention non-interference $A_{ij} = 0$, lower-triangular transition projection, cryptographic capability verification, and True Fused Triton attention) and \textbf{Tier 2 Statistical Monitoring} (speculative multi-layer residual probing over checkpoint coverage $\mathcal{L}_A$). We prove whole-network non-interference via observational equivalence ($X \equiv_L X' \implies \text{Obs}_L(F(X)) = \text{Obs}_L(F(X'))$), demonstrate exact unconstrained transparency ($\Delta \text{PPL} = 0.0000$), evaluate an observed 0.00\% attack success rate across 470 adversarial red-team trials, and eliminate auxiliary $O(N^2)$ policy-mask DRAM allocations ($0.0\text{ MB}$ across all context lengths).
\end{abstract}

\section{Introduction}
Modern LLMs operate on untyped continuous spaces $\mathbb{R}^d$. This architecture fails to provide mathematical guarantees regarding information boundaries.

\begin{definition}[Typed Quad-Tuple Manifold]
Each token hidden state at layer $l$ is represented as an authoritative quad-tuple:
\begin{equation}
h_i^{(l)} = \left( m_i^{(l)}, \sigma_{h,i}^{(l)}, \sigma_{s,i}^{(l)}, \nu_i^{(l)} \right) \in \mathbb{R}^d \times \Sigma_h \times \Sigma_s \times \mathcal{V}
\end{equation}
where $m_i$ is the semantic activation, $\sigma_h \in \Sigma_C \times \Sigma_I \times \Sigma_A \times \Sigma_L$ is the hard trusted policy state, $\sigma_s \in \Sigma_U \times \Sigma_R$ is the soft operational risk state, and $\nu \in \mathcal{V}$ is the value/preference alignment state.
\end{definition}

\section{5x5 Product State Lattice \& Exact Transition Projection}

\subsection{Product State Lattice $\Sigma$}
The product lattice $\Sigma = \Sigma_C \times \Sigma_I \times \Sigma_A \times \Sigma_L \times \Sigma_R$ satisfies complete pairwise orthogonality:
\begin{equation}
\forall (X, Y) \in \{C, I, A, L, R\}^2, X \neq Y \implies \Delta X \implies Y' \equiv Y
\end{equation}
guaranteeing independent enforcement of confidentiality, integrity taint, capability authorizations, license compliance, and operational risk.

\subsection{Exact Transition Projection}
Under the multiplication convention $\sigma' = \sigma V^T \iff \sigma'_j = \sum_i \sigma_i V_{j, i}$, row $j$ represents the destination state and column $i$ represents the source state ($V_{\text{dst}, \text{src}}$). Because a legal transition requires $\text{dst} \ge \text{src}$ ($\text{row} \ge \text{col}$), the legal transition space $\mathcal{T}_\Sigma$ is lower-triangular:
\begin{equation}
\mathcal{P}_{\mathcal{T}_\Sigma}(V) = \text{tril}(V) - \text{diag}(\text{diag}(V)) + \text{diag}(\max(0, \text{diag}(V)))
\end{equation}
rendering unauthorized downward declassifications unrepresentable by construction ($\mathcal{P}(\mathcal{P}(V)) = \mathcal{P}(V)$).

\section{Observational Equivalence \& Non-Interference}

\begin{definition}[Low-Equivalence $\equiv_L$]
Let $L \in \Sigma_h$ denote an observer's clearance. Two input sequences $X, X'$ are low-equivalent ($X \equiv_L X'$) if and only if:
\begin{equation}
\forall t : \sigma_{h,t} \le L \implies X_t = X'_t
\end{equation}
\end{definition}

\begin{definition}[Observer Function $\text{Obs}_L$]
For an output tensor $Y$, $\text{Obs}_L(Y) = \{ Y_t : \sigma_{h,t} \le L \}$.
\end{definition}

\begin{theorem}[Whole-Network Structural Non-Interference]
Let $F$ be an $N$-layer NSA network. Under the structural assumptions of hard state masking, exact lower-triangular projection $V \in \mathcal{T}_\Sigma$, and hard state immutability across continuous layers, for any observer clearance $L$:
\begin{equation}
X \equiv_L X' \implies \text{Obs}_L(F(X)) = \text{Obs}_L(F(X'))
\end{equation}
\end{theorem}

\begin{proof}
For any layer $l$ and query position $i$ with $\sigma_{h,i} \le L$, attention weights $A_{ij}^{(l)} = 0$ for all $j$ where $\sigma_{h,j} \not\le \sigma_{h,i}$. Hence, the output activation $v_{\text{out}, i}^{(l)}$ depends exclusively on coordinates with $\sigma_{h,j} \le L$, which are identical between $X$ and $X'$. By induction on depth, $\text{Obs}_L(F(X)) = \text{Obs}_L(F(X'))$.
\end{proof}
